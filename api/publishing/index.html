    Publishing¶

    The published guest image: vm_artifact is the artifact's format, publish_qcow pushes one and publish_prune retires old tags.

    qemu_tool.vm_artifact ¶

    The published qcow2 artifact's format, shared by the puller and the pusher.

    gen_vm pulls it (--backing-image) and publish_qcow pushes it. Both read the names from here so that neither can drift from the other.

    The main artifact is one zstd layer holding a standalone qcow2. A referrer attached to it carries vm-info.json and the guest keypair, so a consumer can fetch credentials without pulling the disk.

    qemu_tool.publish_qcow ¶

    Publish a baked guest qcow2 as an OCI artifact.

    The other half of gen_vm's --backing-image: what this pushes is what that pulls. See vm_artifact.py for the format.

    What is published is the file named by --image and nothing derived from a boot of it. A caller that tests the guest before publishing does so on an overlay, so module loads, test output and anything specific to one host's devices stay out of the artifact.

    def image_facts(image: Path) -> dict:
    """Format and sizes of *image*, refusing one that is not standalone."""
    info = _run_json(["qemu-img", "info", "--output=json", str(image)])
    if info.get("format") != "qcow2":
        sys.exit(f"Error: {image} is {info.get('format')}, not qcow2.")
    if info.get("backing-filename"):
        sys.exit(
            f"Error: {image} is an overlay on {info['backing-filename']}. "
            "Publish the baked base, not an overlay a guest has run on."
        )
    return info

    def build_vm_info(args: argparse.Namespace, facts: dict, qcow2_name: str,
                  size_bytes: int, fingerprint: str, created: str) -> dict:
    """The vm-info.json a consumer reads instead of booting the image.

    The field names are a contract with every consumer of the published
    guests; add to them, never rename. The guest's own stamp is carried
    whole under "guest", so the pins it records are stated once, there.
    """
    guest = json.loads(args.guest_stamp.read_text()) if args.guest_stamp else {}
    return {
        "schema_version": VM_INFO_SCHEMA_VERSION,
        "vm_name": args.vm_name,
        "flavour": args.flavour,
        "username": args.username,
        "password": args.password,
        "image_path": f"/output/{qcow2_name}",
        "image_format": "qcow2",
        "image_size_bytes": size_bytes,
        "image_virtual_size_bytes": facts["virtual-size"],
        "release": args.release,
        "architecture": args.architecture,
        "kernel_release": guest.get("booted_kernel", ""),
        "backing_file": False,
        "ssh_keys": {
            "private_key_path": f"/output/{VM_KEY_NAME}",
            "public_key_path": f"/output/{VM_PUB_NAME}",
            "fingerprint": fingerprint,
        },
        "rocm": {
            "path": guest.get("rocm_path", ""),
            "version": guest.get("amdrocm_runtime_dev_version", ""),
        },
        "provisioning": {
            "vm_playbook": args.playbook,
            "kernel_ref": guest.get("ionic_kernel_pin", ""),
            "kernel_debs": " ".join(guest.get("kernel_debs", [])),
            "ernic_commit": guest.get("ernic_source_pin", ""),
            "sdxi_commit": guest.get("sdxi_source_ref", ""),
            "amdgpu_dkms_version": guest.get("amdgpu_dkms_version", ""),
            "unattended_upgrades": "purged",
        },
        "guest": guest,
        "build_info": {
            "qemu_minimal_commit": args.commit,
            "build_timestamp": created,
        },
    }

    def build_annotations(args: argparse.Namespace, info: dict, version: str) -> dict:
    """Annotations for the main artifact.

    They let `oras manifest fetch` answer "what is this" without a multi-GB
    pull. The password is in vm-info.json, which a consumer fetches on
    purpose, and never here.
    """
    annotations = {
        "org.opencontainers.image.title": f"{info['vm_name']}.qcow2.zst",
        "org.opencontainers.image.description": (
            f"{info['flavour']} guest qcow2 ({info['release']}, "
            f"{info['architecture']}, kernel {info['kernel_release']})"
        ),
        "org.opencontainers.image.version": version,
        "org.opencontainers.image.revision": args.commit,
        "org.opencontainers.image.created": info["build_info"]["build_timestamp"],
        "org.opencontainers.image.source": args.source,
        "vm.name": info["vm_name"],
        "vm.flavour": info["flavour"],
        "vm.release": info["release"],
        "vm.architecture": info["architecture"],
        "vm.kernel_release": info["kernel_release"],
        "vm.username": info["username"],
        "vm.image_size_bytes": info["image_size_bytes"],
        "vm.playbook": info["provisioning"]["vm_playbook"],
        "vm.qemu_minimal_commit": args.commit,
    }
    return {k: str(v) for k, v in annotations.items() if v not in ("", None)}

    def resolve(args: argparse.Namespace) -> None:
    """Fill the flags left unset from the --xml layers, then the defaults.

    Only an explicit --xml is read. An ambient fleet.xml must not decide
    what goes to a registry.
    """
    fleet = FleetXml.parse_layers(FleetXml.find(args.xml)) if args.xml else None
    have = fleet or FleetConfig()

    def fill(attr: str, from_xml: str, default: str | None = None) -> None:
        if getattr(args, attr) is None:
            setattr(args, attr, from_xml or default)

    fill("ref", have.publish_ref)
    fill("vm_name", have.bake_name)
    fill("flavour", have.publish_flavour, "rocm-golden")
    fill("username", have.bake_username, "ubuntu")
    fill("password", have.bake_password, "password")
    fill("release", have.bake_release, "resolute")
    fill("playbook", have.bake_playbook, "")

    if args.image is None and fleet is not None and args.vm_name:
        # The name gen-vm gives the image it bakes.
        args.image = (Path(fleet.vm_images_dir).expanduser()
                      / f"{args.vm_name}-backing.qcow2")
    # --key is not filled in from a layer. <vm-ssh-key-file> is a fleet's
    # own login key, usually somebody's ~/.ssh one, and this key is
    # published; run() takes the image's generated key from its sidecar.

    missing = [
        hint for attr, hint in (
            ("ref", "--ref, or <publish-ref> in an --xml layer"),
            ("vm_name", "--vm-name, or <bake-name> in an --xml layer"),
            ("image", "--image, or --xml with <bake-name> (or --vm-name) to "
                      "find it under <vm-images-dir>"),
        ) if not getattr(args, attr)
    ]
    if missing:
        sys.exit("Error: publish-qcow needs:\n  " + "\n  ".join(missing))

    def run(args: argparse.Namespace) -> None:
    resolve(args)
    tools = ["qemu-img", "zstd", "ssh-keygen"] + ([] if args.dry_run else ["oras"])
    for tool in tools:
        if shutil.which(tool) is None:
            sys.exit(f"Error: publish-qcow needs '{tool}', which is not on PATH.")

    image = args.image.resolve()
    if not image.is_file():
        sys.exit(f"Error: {image} does not exist.")
    pulled = image.with_name(image.name + ".image-ref")
    if pulled.is_file():
        sys.exit(
            f"Error: {image} was pulled from {pulled.read_text().strip()} by "
            "gen-vm, not baked here;\n"
            "       publishing it would republish that image as a new bake. "
            "Bake with 'gen-vm --bake' first."
        )
    # Before the key is looked for: an overlay shares its backing image's
    # record, and is refused for what it is.
    facts = image_facts(image)
    sidecar = _sidecar(image)
    private_key = (args.key.resolve() if args.key
                   else _generated_key(sidecar, image))
    public_key = private_key.with_name(private_key.name + ".pub")
    for key in (private_key, public_key):
        if not key.is_file():
            sys.exit(f"Error: {key} does not exist.")
    _check_baked_key(sidecar, public_key)
    _check_pair(private_key, public_key)
    if args.guest_stamp and not args.guest_stamp.is_file():
        sys.exit(f"Error: {args.guest_stamp} does not exist.")
    if not args.tag:
        sys.exit("Error: at least one --tag is required.")

    created = datetime.datetime.now(datetime.timezone.utc).strftime(
        "%Y-%m-%dT%H:%M:%SZ")
    qcow2_name = f"{args.vm_name}.qcow2"

    # Beside the image by default: /tmp is often a tmpfs, and this holds a
    # full copy of the disk.
    work_parent = args.work_dir or image.parent
    with tempfile.TemporaryDirectory(dir=work_parent, prefix="publish-qcow-") as tmp:
        work = Path(tmp)
        qcow2 = work / qcow2_name

        # A fresh copy, not the file itself: convert drops the clusters the
        # bake freed and writes none compressed. zstd does the compression;
        # compressed clusters inside the image slow every consumer's reads.
        print(f"Converting {image.name} to a standalone qcow2")
        subprocess.run(
            ["qemu-img", "convert", "-O", "qcow2", str(image), str(qcow2)],
            check=True,
        )
        size_bytes = qcow2.stat().st_size

        shutil.copyfile(private_key, work / VM_KEY_NAME)
        (work / VM_KEY_NAME).chmod(0o600)
        shutil.copyfile(public_key, work / VM_PUB_NAME)

        info = build_vm_info(
            args, facts, qcow2_name, size_bytes,
            key_fingerprint(public_key), created,
        )
        (work / VM_INFO_NAME).write_text(json.dumps(info, indent=2) + "\n")

        print(f"Compressing {qcow2_name} ({size_bytes} bytes)")
        subprocess.run(
            ["zstd", *ZSTD_ARGS, "--rm", "-q", "-f", str(qcow2)], check=True,
        )
        blob = qcow2.with_name(qcow2_name + ".zst")
        print(f"Compressed to {blob.stat().st_size} bytes")

        annotations = build_annotations(args, info, args.tag[0])
        referrer = {k: annotations[k] for k in _REFERRER_ANNOTATIONS
                    if k in annotations}
        first = f"{args.ref}:{args.tag[0]}"

        if args.dry_run:
            for tag in args.tag:
                print(f"Would push {args.ref}:{tag}")
            print(f"Would attach {VM_INFO_NAME}, {VM_KEY_NAME}, {VM_PUB_NAME}")
            for key, value in annotations.items():
                print(f"  {key} = {value}")
            if args.keep:
                kept = args.keep.resolve()
                kept.mkdir(parents=True, exist_ok=True)
                for name in (blob.name, VM_INFO_NAME):
                    shutil.move(str(work / name), str(kept / name))
                print(f"Kept {blob.name} and {VM_INFO_NAME} in {kept}")
            return

        # One push carries every tag; the referrer attaches to the first.
        subprocess.run(
            ["oras", "push", "--artifact-type", VM_ARTIFACT_TYPE,
             *_annotation_args(annotations),
             f"{args.ref}:{','.join(args.tag)}",
             f"{blob.name}:{VM_LAYER_MEDIA_TYPE}"],
            cwd=work, check=True,
        )
        subprocess.run(
            ["oras", "attach", "--artifact-type", VM_INFO_ARTIFACT_TYPE,
             *_annotation_args(referrer), first,
             f"{VM_INFO_NAME}:application/json",
             f"{VM_KEY_NAME}:{VM_SSH_KEY_MEDIA_TYPE}",
             f"{VM_PUB_NAME}:{VM_SSH_KEY_MEDIA_TYPE}"],
            cwd=work, check=True,
        )

    digest = _run_json(
        ["oras", "manifest", "fetch", "--descriptor", first]).get("digest", "")
    for tag in args.tag:
        print(f"Pushed {args.ref}:{tag}")
    print(f"Digest {args.ref}@{digest}")
    if args.digest_file:
        args.digest_file.write_text(f"{args.ref}@{digest}\n")

    qemu_tool.publish_prune ¶

    Retention for the published guest image's repository on Docker Hub.

    publish-qcow pushes each image under a dated tag, "<YYYYMMDD>.g<sha7>", and moves two rolling tags onto it. Nothing ever removed a dated tag, so the repository grew by a multi-gigabyte image per publish. This lists the tags, decides which dated ones the policy no longer keeps, prints that plan, and deletes them only when told to.

    A tag is a name, and names are all this removes. The manifest a deleted tag pointed at, its blobs, and its referrer (vm-info.json and the image's keypair) are the registry's to keep or collect. Docker Hub serves the OCI referrers API, so oras attaches the referrer to the manifest and there is no "sha256-<hex>" fallback tag there to delete; one that did turn up is not a dated tag and is kept like any other.

    The plan is a pure function of the listing (plan()), so that everything that decides what is deleted can be tested without a registry.

    Docker Hub only: tags are deleted through the Hub API, which other registries do not have. The endpoints, read from Docker's published OpenAPI description and from the live API:

    POST   /v2/auth/token                         {identifier, secret}
                                              -> {access_token}, a bearer
GET    /v2/namespaces/<ns>/repositories/<repo>/tags?page=N&page_size=100
                                              -> {count, next, results[]}
DELETE /v2/repositories/<ns>/<repo>/tags/<tag>/  -> 204

    The DELETE is the one the Hub's own web interface sends. It is not in the published description, which documents only GET and HEAD for a tag.

    if "@" in ref or ":" in ref.rsplit("/", 1)[-1]:
        raise PruneError(
            f"{ref!r} carries a tag or a digest; name the repository alone, "
            "as <publish-ref> does.")
    parts = ref.split("/")
    if len(parts) != 3 or parts[0] not in HUB_REGISTRIES:
        raise PruneError(
            f"{ref!r} is not a Docker Hub repository. Tags are deleted "
            "through the Docker Hub API, which no other registry has, so "
            "the reference must read docker.io/<namespace>/<repository>.")
    namespace, repository = parts[1], parts[2]
    for part in (namespace, repository):
        if not re.fullmatch(r"[a-z0-9]+(?:[._-]+[a-z0-9]+)*", part):
            raise PruneError(f"{ref!r} is not a valid repository name.")
    return namespace, repository

    def parse_time(text: object) -> datetime.datetime | None:
    """An API timestamp as an aware datetime; None for anything else.

    The Hub writes between zero and nine fractional digits, which
    fromisoformat does not accept on every supported Python. A time from
    before the Hub existed is not a push time, and is None as well.
    """
    if not isinstance(text, str):
        return None
    found = re.fullmatch(
        r"(\d{4})-(\d\d)-(\d\d)T(\d\d):(\d\d):(\d\d)(?:\.(\d+))?Z", text,
        re.ASCII)
    if not found:
        return None
    *whole, fraction = found.groups()
    micro = int((fraction or "0")[:6].ljust(6, "0"))
    try:
        when = datetime.datetime(*map(int, whole), micro,
                                 tzinfo=datetime.timezone.utc)
    except ValueError:
        return None
    return when if when.year >= EARLIEST_YEAR else None

    def plan(tags: Iterable[Tag], *, keep: int, days: int,
         now: datetime.datetime) -> Plan:
    """Which tags the policy keeps and which it deletes.

    Kept: the *keep* newest dated tags, every dated tag pushed within *days*
    days of *now*, and every tag that is not a dated one. A dated tag whose
    push time the listing did not give is kept, and is not one of the
    *keep*: its age is unknown.
    """
    if keep < 1:
        raise PruneError("--keep must be at least 1: the newest published "
                         "image is never deleted.")
    if days < 0:
        raise PruneError("--days must not be negative.")
    tags = list(tags)
    names = [tag.name for tag in tags]
    if len(set(names)) != len(names):
        raise PruneError("the tag listing names a tag twice; not acting on "
                         "a listing that changed while it was read.")

    cutoff = now - datetime.timedelta(days=days)
    dated = [tag for tag in tags if DATED.fullmatch(tag.name)]
    timed = sorted((tag for tag in dated if tag.pushed is not None),
                   key=lambda tag: (tag.pushed, tag.name), reverse=True)
    kept: list[tuple[Tag, str]] = []
    doomed: list[tuple[Tag, str]] = []
    for rank, tag in enumerate(timed, start=1):
        age = (now - tag.pushed).days
        if rank <= keep:
            kept.append((tag, f"dated, newest {rank} of {keep} kept"))
        elif tag.pushed >= cutoff:
            kept.append((tag, f"dated, pushed {age} days ago, within {days}"))
        else:
            doomed.append((tag, f"dated, pushed {age} days ago, "
                                f"not among the {keep} newest"))
    for tag in dated:
        if tag.pushed is None:
            kept.append((tag, "dated, but the listing gives no push time"))

    for tag in tags:
        if DATED.fullmatch(tag.name):
            continue
        if tag.name == "latest":
            kept.append((tag, "latest, always kept"))
        elif ROLLING.fullmatch(tag.name):
            kept.append((tag, "qm.<sha7>, always kept"))
        else:
            kept.append((tag, "not a dated tag, never a candidate"))
    return Plan(keep=tuple(kept), delete=tuple(doomed))

    def http_transport(method: str, url: str, headers: dict,
                   body: bytes | None) -> tuple[int, object]:
    request = urllib.request.Request(url, data=body, method=method,
                                     headers=headers)
    try:
        with _OPENER.open(request, timeout=60) as response:
            status, raw = response.status, response.read()
    except urllib.error.HTTPError as error:
        with error:
            try:
                status, raw = error.code, error.read()
            except (OSError, http.client.HTTPException):
                status, raw = error.code, b""
    # HTTPException is not an OSError: a truncated or garbled response.
    except (urllib.error.URLError, OSError,
            http.client.HTTPException) as error:
        raise PruneError(
            f"{method} {_path(url)}: {type(error).__name__}: {error}"
        ) from None
    try:
        return status, json.loads(raw) if raw.strip() else None
    except ValueError:
        return status, None

    def login(transport: Transport, user: str, secret: str) -> str:
    """A short-lived bearer token for *user*. The secret is never echoed."""
    status, payload = transport(
        "POST", f"{HUB}/v2/auth/token",
        {"Content-Type": "application/json"},
        json.dumps({"identifier": user, "secret": secret}).encode())
    token = payload.get("access_token") if isinstance(payload, dict) else None

            f"{TOKEN_ENV} (HTTP {status}).")

    def list_tags(transport: Transport, namespace: str, repository: str,
              token: str | None) -> list[Tag]:
    """Every tag of the repository, or PruneError: never part of them."""
    headers = {"Accept": "application/json"}
    if token:
        headers["Authorization"] = f"Bearer {token}"
    base = (f"{HUB}/v2/namespaces/{namespace}/repositories/{repository}"
            f"/tags?page_size={PAGE_SIZE}")
    results: list[dict] = []
    count = None
    for page in range(1, MAX_PAGES + 1):
        # Built here, not taken from "next", and the transport follows no
        # redirect: the token goes nowhere a response names.
        url = f"{base}&page={page}"
        status, payload = transport("GET", url, headers, None)
        if status != 200:
            raise PruneError(
                f"listing {namespace}/{repository} failed on page {page} "
                f"(HTTP {status}{_detail(payload)}). Nothing was deleted.")
        if (not isinstance(payload, dict)
                or not isinstance(payload.get("results"), list)
                or not isinstance(payload.get("count"), int)):
            raise PruneError(
                f"page {page} of the tag listing is not the shape the Hub "
                "API documents. Nothing was deleted.")
        if count is None:
            count = payload["count"]
        elif payload["count"] != count:
            raise PruneError(
                "the repository's tag count changed while it was listed "
                f"({count}, then {payload['count']}). Nothing was deleted; "
                "run it again.")
        results += payload["results"]
        if not payload.get("next"):
            break
        if not payload["results"]:
            raise PruneError(
                f"page {page} of the tag listing is empty but names a next "
                "page. Nothing was deleted.")
    else:
        raise PruneError(
            f"the tag listing did not end within {MAX_PAGES} pages. "
            "Nothing was deleted.")
    if len(results) != count:
        raise PruneError(
            f"the tag listing is incomplete: {len(results)} of {count} "
            "tags. Nothing was deleted.")
    return tags_from_results(results)

    def render(ref: str, result: Plan, *, keep: int, days: int) -> str:
    lines = [f"{ref}: keep the {keep} newest dated tags and any pushed "
             f"within {days} days; latest and qm.* always."]
    width = max((len(tag.name) for tag, _ in result.keep + result.delete),
                default=0)
    # A tag may be 128 characters; do not let one set the column.
    width = min(width, 24)
    for verb, group in (("keep  ", result.keep), ("DELETE", result.delete)):
        for tag, why in group:
            lines.append(f"  {verb} {tag.name:<{width}}  {_when(tag)}  {why}")
    lines.append(f"{len(result.keep)} kept, {len(result.delete)} to delete.")
    return "\n".join(lines)
