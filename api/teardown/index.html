    Teardown¶

    shutdown powers guests off through the guest agent, and down_all finds the stacks a caller owns and takes them down.

    qemu_tool.shutdown ¶

    Shut guests down cleanly before their containers are stopped.

    docker compose down sends SIGTERM to PID 1, and run_vm execs QEMU, so PID 1 IS qemu-system-x86_64. QEMU exits on SIGTERM; it does not ACPI-shutdown the guest. Every teardown is therefore the equivalent of pulling the power cord, and the guest's filesystem is left needing journal replay -- or, observed on a 32-guest fleet, an aborted journal and a root remounted read-only.

    Nothing in the stack had a place to hook this: there is no shim to trap the signal, and raising stop_grace_period alone only makes Docker wait longer before killing a guest that was never asked to stop. So the request has to come from outside, over the guest agent, before the containers are stopped.

    QGA rather than QMP because the fleet already has it: run_vm wires a virtserialport on every guest unconditionally, while QMP is opt-in and off by default (qmp_available: false on a fleet guest).

    def run(vm_name: str | None, timeout: int = 120, dry_run: bool = False,
        stack_filter: str | None = None) -> None:
    try:
        vms = _collect_vms()
    except OSError as exc:
        sys.exit(f"Error: cannot read /proc: {exc}")

    if stack_filter is not None:
        prefix = f"{stack_filter}-qemu-"
        vms = [v for v in vms if (v.get("container") or "").startswith(prefix)]
        if not vms:
            print(f"No running VMs in stack {stack_filter!r}.")
            return
    elif vm_name is not None:
        vms = [v for v in vms if v["name"] == vm_name]
        if not vms:
            sys.exit(f"Error: no running VM named {vm_name!r}")
    if not vms:
        print("No running VMs.")
        return

    if dry_run:
        for vm in vms:
            print(f"would shut down {vm['name']} via {vm.get('ga_socket')}")
        return

    print(f"Requesting shutdown of {len(vms)} VM(s)...")
    # Concurrent, because a fleet of 32 at ~1 s each is otherwise half a
    # minute of waiting before the first guest has even begun to stop.
    workers = min(8, len(vms))
    with ThreadPoolExecutor(max_workers=workers) as pool:
        results = list(pool.map(_request_shutdown, vms))

    asked = set()
    for name, ok, err in results:
        if ok:
            asked.add(name)
        else:
            print(f"  {name}: shutdown not requested ({err})", file=sys.stderr)

    if not asked:
        sys.exit(
            "Error: no guest accepted a shutdown request.\n"
            "       Stopping the containers now would hard-kill them; check "
            "the agent is running."
        )

    deadline = time.monotonic() + timeout
    while time.monotonic() < deadline:
        left = _still_running(asked)
        if not left:
            print(f"All {len(asked)} guest(s) powered off.")
            return
        time.sleep(2)

    left = _still_running(asked)
    # Not an error: the caller's next step is to stop the containers anyway,
    # and saying which guests did not finish is more useful than refusing.
    print(
        f"Warning: {len(left)} guest(s) still running after {timeout}s: "
        + ", ".join(sorted(left))
        + "\n         Stopping them now will not be clean.",
        file=sys.stderr,
    )

    qemu_tool.down_all ¶

    qemu-tool down-all -- tear down every qemu-tool stack you own.

    Two questions have to be answered soundly before anything is destroyed, and compose records the evidence for both on every container it starts:

    • Is this a qemu-tool stack? com.docker.compose.project.config_files names the compose file, and every stack this tool runs is generated, so that file carries the gen-compose provenance header. That is not a guess, and the alternative -- matching service names like qemu-1 -- would happily tear down somebody's unrelated project that happened to have a service called qemu-1.

    • Is it yours? .Config.User is the uid the container actually runs as, which is exactly what the CONTAINER_USER work put there. A project is yours if any of its containers runs as your uid.

    The second has a limitation worth stating rather than discovering: a stack started BEFORE the containers ran as the caller has every container as root, so to a non-root caller it is root's and is skipped. That is the correct answer -- you cannot docker stop your way out of not owning it -- but it is not an obvious one, so skipped stacks are listed with the reason.

    Destroying things is confirmed first, every time, unless --yes.

    @dataclass
class Stack:
    project: str
    config_file: Path | None
    working_dir: Path | None
    env_files: list[Path]
    containers: list[str] = field(default_factory=list)
    uids: set[str] = field(default_factory=set)
    # Another compose project on this host uses the same project name.
    shared_name: bool = False

    def owned_by(self, uid: int) -> bool:
        """Does any container run as this uid?

        Any, not all: a stack legitimately mixes uids -- the stats exporters
        are pinned to 65534 and tapsetup/alloy stay root -- so requiring
        all of them would find nothing.
        """
        return any(u.split(":")[0] == str(uid) for u in self.uids if u)

    def discover() -> list[Stack]:
    """Every running qemu-tool stack on this host, owned by anyone."""
    # Keyed on the compose file as well as the project name: the name is the
    # directory basename, which any unrelated project can share.
    by_project: dict[tuple[str, str], Stack] = {}
    for rec in _inspect_running():
        labels = (rec.get("Config") or {}).get("Labels") or {}
        project = labels.get("com.docker.compose.project")
        if not project:
            continue
        raw_cfg = (labels.get("com.docker.compose.project.config_files")
                   or "").split(",")[0]
        key = (project, raw_cfg)
        if key not in by_project:
            raw_wd = labels.get("com.docker.compose.project.working_dir") or ""
            raw_env = labels.get(
                "com.docker.compose.project.environment_file") or ""
            by_project[key] = Stack(
                project=project,
                config_file=Path(raw_cfg) if raw_cfg else None,
                working_dir=Path(raw_wd) if raw_wd else None,
                env_files=[Path(p) for p in raw_env.split(",") if p],
            )
        stack = by_project[key]
        stack.containers.append((rec.get("Name") or "").lstrip("/"))
        stack.uids.add((rec.get("Config") or {}).get("User") or "")

    names = [project for project, _ in by_project]
    for (project, _), stack in by_project.items():
        stack.shared_name = names.count(project) > 1
    return sorted(
        (s for s in by_project.values() if _is_qemu_tool_stack(s.config_file)),
        key=lambda s: (s.project, str(s.config_file)),
    )

    def teardown_one(project: str, dry_run: bool = False,
                 assume_yes: bool = False) -> None:
    """Tear down one named stack, identified by its compose project name."""
    stacks = discover()
    match = next((s for s in stacks if s.project == project), None)
    if match is None:
        sys.exit(f"Error: no running qemu-tool stack named {project!r}.")
    uid = os.getuid()
    print(f"Will tear down stack:")
    print(_describe(match, uid))
    if dry_run:
        print(f"\n--dry-run: nothing was stopped. The command would be:")
        print(f"  {' '.join(_teardown_cmd(match))}")
        return
    if not assume_yes:
        total = len(match.containers)
        try:
            reply = input(
                f"\nStop and remove {total} container(s) in "
                f"{project!r}? [y/N]: ").strip().lower()
        except EOFError:
            reply = ""
        if reply not in ("y", "yes"):
            print("Nothing was stopped.")
            return
    result = subprocess.run(_teardown_cmd(match), env=_teardown_env(match))
    if result.returncode != 0:
        sys.exit(f"Stack {project!r} did not tear down (exit {result.returncode}).")
    print(f"\nStack {project!r} torn down.")

    def run(assume_yes: bool = False, dry_run: bool = False,
        all_users: bool = False) -> None:
    uid = os.getuid()
    stacks = discover()
    if not stacks:
        print("No running qemu-tool stacks found.")
        return

    if all_users and uid != 0:
        sys.exit(
            "Error: --all-users needs root -- another user's containers "
            "cannot be stopped\n"
            "       without it. Re-run under sudo."
        )
    # Root owns the machine, so "mine" is everything; that is what makes
    # `sudo qemu-tool down-all` the system-wide form the way it reads.
    take_all = all_users or uid == 0
    mine = [s for s in stacks if take_all or s.owned_by(uid)]
    skipped = [s for s in stacks if s not in mine]

    if mine:
        print(f"Will tear down {len(mine)} stack(s):")
        for stack in mine:
            print(_describe(stack, uid))
    if skipped:
        print(f"\nSkipping {len(skipped)} stack(s) you do not own:")
        for stack in skipped:
            print(_describe(stack, uid))
        print("\n  A stack started before the containers ran as the caller "
              "has every\n"
              "  container as root, so it reads as root's. Re-run under sudo "
              "to include it.")
    if not mine:
        return

    if dry_run:
        print("\n--dry-run: nothing was stopped. The commands would be:")
        for stack in mine:
            print(f"  {' '.join(_teardown_cmd(stack))}")
        return

    if not assume_yes:
        # Always, and never defaulting to yes. This removes containers; a
        # bare Enter must not be the thing that does it.
        total = sum(len(s.containers) for s in mine)
        try:
            reply = input(
                f"\nStop and remove {total} containers "
                f"across {len(mine)} stack(s)? [y/N]: ").strip().lower()
        except EOFError:
            reply = ""
        if reply not in ("y", "yes"):
            print("Nothing was stopped.")
            return

    failed = 0
    for stack in mine:
        print(f"\n==> {stack.project}")
        result = subprocess.run(_teardown_cmd(stack), env=_teardown_env(stack))
        if result.returncode != 0:
            failed += 1
            print(f"    failed (exit {result.returncode}); "
                  f"remove by name with: docker rm -f "
                  f"{' '.join(stack.containers)}", file=sys.stderr)

    if failed:
        sys.exit(f"\n{failed} of {len(mine)} stack(s) did not tear down.")
    print(f"\n{len(mine)} stack(s) torn down.")
